For this question my approach was to just brute force and check directions left, right, top and bottom for each cell in the input matrix to check for special cells. 

```python
class Solution:
    def numSpecial(self, mat: List[List[int]]) -> int:
        ROWS = len(mat)
        COLS = len(mat[0])
        special = 0
        
        for r in range(ROWS):
            for c in range(COLS):
                if mat[r][c] == 0:
                    continue

                valid = True

                for i in range(c + 1, COLS): # check right
                    if mat[r][i] == 1:
                        valid = False
                        break

                if valid:
                    for j in range(r + 1, ROWS): # check bottom
                        if mat[j][c] == 1:
                            valid = False
                            break

                if valid:
                    for k in range(c): # check left
                        if mat[r][k] == 1:
                            valid = False
                            break

                if valid:
                    for l in range(r): # check top
                        if mat[l][c] == 1:
                            valid = False
                            break

                if valid:
                    special += 1

        return special
```

> Time Complexity : `O(r * c * (r + c))`
> 
> Space Complexity : `O(1)`
