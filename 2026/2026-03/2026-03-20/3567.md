Just brute force it

```python
class Solution:
    def minAbsDiff(self, grid: List[List[int]], k: int) -> List[List[int]]:
        ROWS = len(grid) 
        COLS = len(grid[0])
        res = [[0] * (COLS - k + 1) for _ in range(ROWS - k + 1)]

        for r in range(ROWS - k + 1):
            for c in range(COLS - k + 1):
                submatrix = []
                for i in range(r, r + k):
                    for j in range(c, c + k):
                        submatrix.append(grid[i][j])
                
                submatrix.sort()
                if len(submatrix) == 1:
                    res[r][c] = 0
                else:
                    minDiff = inf
                    prev = submatrix[0]
                    for l in range(1, len(submatrix)):
                        if submatrix[l] == prev:
                            continue
                        minDiff = min(minDiff, abs(prev - submatrix[l]))
                        prev = submatrix[l]
                        
                    if minDiff == inf:
                        res[r][c] = 0
                    else:
                        res[r][c] = minDiff

        return res
```

> Time Complexity : `O((m - k) * (n - k) * k^2 * log k)`
> 
> Space Complexity : `O(k^2)`
