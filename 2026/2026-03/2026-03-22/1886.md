For this question we can simulate all possible rotation. For each rotation we can check if the current rotation is equal to the target matrix, if it is we will return true. If we compare all possible rotations and none return true then we will return false.

```python
class Solution:
    def findRotation(self, mat: List[List[int]], target: List[List[int]]) -> bool:
        ROWS = len(mat) 
        COLS = len(mat[0])

        # without rotation
        if mat == target:
            return True

        # rotate once
        one = []
        for c in range(COLS):
            row = []
            for r in range(ROWS - 1, -1, -1):
                row.append(mat[r][c])
            one.append(row)
        if one == target:
            return True

        # rotate twice
        two = []
        for r in range(COLS - 1, -1, -1):
            row = []
            for c in range(ROWS - 1, -1, -1):
                row.append(mat[r][c])
            two.append(row)
        if two == target:
            return True

        # rotate thrice
        three = []
        for c in range(COLS - 1, -1, -1):
            row = []
            for r in range(ROWS):
                row.append(mat[r][c])
            three.append(row)
        if three == target:
            return True

        return False
```

> Time Complexity : `O(n^2)`
> 
> Space Complexity : `O(n)`
