```python
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def nodesBetweenCriticalPoints(self, head: Optional[ListNode]) -> List[int]:
        critical = []

        prev = head
        curr = prev.next
        i = 1

        while curr and curr.next:
            if curr.val > prev.val and curr.val > curr.next.val:
                critical.append(i)
            elif curr.val < prev.val and curr.val < curr.next.val:
                critical.append(i)

            temp = curr
            prev = curr
            curr = temp.next
            i += 1

        if len(critical) < 2:
            return [-1, -1]

        max_distance = critical[-1] - critical[0]
        min_distance = inf

        for i in range(len(critical) - 1):
            min_distance = min(min_distance, critical[i + 1] - critical[i])

        return [min_distance, max_distance]
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(n)`
