```python
class Solution:
    def countDaysTogether(self, arriveAlice: str, leaveAlice: str, arriveBob: str, leaveBob: str) -> int:
        # key ideas:
        # have a way to parse the date format MM-DD
        # once we have parsed MM-DD for arrival and leave time
        # for alice and bob we can check for the days of overlap
        # we can iterate through days of month using
        # days = [0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]

        def parse_date(date: str) -> tuple:
            return (int(date[:2]), int(date[3:]))

        arriveAlice = parse_date(arriveAlice)
        leaveAlice = parse_date(leaveAlice)
        arriveBob = parse_date(arriveBob)
        leaveBob = parse_date(leaveBob)
        days = [0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
        res = 0

        for i in range(1, 12 + 1):
            for j in range(1, days[i] + 1):
                if ((i, j) >= arriveAlice and (i, j) <= leaveAlice and
                    (i, j) >= arriveBob and (i, j) <= leaveBob):
                    res += 1

        return res
```

> Time Complexity : `O(1)`
>
> Space Complexity : `O(1)`
