```python
class Solution:
    def minSwaps(self, nums: List[int]) -> int:
        pairs = []
        for num in nums:
            total = 0
            for digit in str(num):
                total += int(digit)
            pairs.append((total, num))  # (digit sum, value)

        sorted_pairs = sorted(pairs)
        sorted_val_to_index = {}
        for i in range(len(sorted_pairs)):
            sorted_val_to_index[sorted_pairs[i][1]] = i

        n = len(nums)
        visited = [False] * n
        res = 0
        for i in range(n):
            if visited[i] or sorted_val_to_index[nums[i]] == i:
                visited[i] = True
                continue

            # walk the cycle starting at i
            cycle_len = 0
            j = i
            while not visited[j]:
                visited[j] = True
                j = sorted_val_to_index[nums[j]]
                cycle_len += 1

            res += cycle_len - 1

        return res
```

> Time Complexity : `O(n * log n)`
>
> Space Complexity : `O(n)`
