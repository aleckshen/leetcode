```python
class Solution:
    def isValid(self, s: str) -> bool:
        stack = []
        pairs = {')': '(', '}': '{', ']': '['}

        for br in s:
            if br not in pairs:
                stack.append(br)
            else:
                if stack and stack[-1] == pairs[br]:
                    stack.pop()
                else:
                    return False

        return False if stack else True
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(n)`
