```python
import sys

class Solution:
    def findSubtreeSizes(self, parent: List[int], s: str) -> List[int]:
        n = len(parent)
        children = [[] for _ in range(n)]
        for i in range(1, n):
            children[parent[i]].append(i)

        ans = [1] * n
        last = [-1] * 26

        def dfs(x: int, p: int) -> None:
            c = ord(s[x]) - 97
            new_p = last[c] if last[c] != -1 else p
            old = last[c]
            last[c] = x

            for ch in children[x]:
                dfs(ch, x)

            last[c] = old
            if new_p != -1:
                ans[new_p] += ans[x]

        dfs(0, -1)
        return ans
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(n)`
