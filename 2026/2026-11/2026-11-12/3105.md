```python
class Solution:
    def longestMonotonicSubarray(self, nums: List[int]) -> int:
        up = 1
        down = 1
        res = 1

        for i in range(1, len(nums)):
            if nums[i] > nums[i - 1]:
                up +=
            else:
                up = 1

            if nums[i] < nums[i - 1]:
                down += 1
            else:
                down = 1
            res = max(res, up, down)

        return res
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(1)`
