```python
class Solution:
    def countNonDecreasingSubarrays(self, nums: List[int], k: int) -> int:
        n = len(nums)
        dq = deque()
        cost = 0
        r = n - 1
        res = 0

        for i in range(n - 1, -1, -1):
            while dq and nums[dq[0]] <= nums[i]:
                top = dq.popleft()
                end = dq[0] if dq else r + 1
                cost += (nums[i] - nums[top]) * (end - top)
            dq.appendleft(i)

            while cost > k:
                mx = dq[-1]
                cost -= nums[mx] - nums[r]
                if mx == r:
                    dq.pop()
                r -= 1

            res += r - i + 1

        return res
```

> Time Complexity : `O(n * log n)`
>
> Space Complexity : `O(n)`
