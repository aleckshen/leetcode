```python
class Solution:
    def firstStableIndex(self, nums: list[int], k: int) -> int:
        n = len(nums)
        max_val = []

        curr_max = -inf
        for i in range(n):
            curr_max = max(curr_max, nums[i])
            max_val.append(curr_max)

        res = inf
        curr_min = inf
        for i in range(n - 1, -1, -1):
            curr_min = min(curr_min, nums[i])
            score = max_val[i] - curr_min
            if score <= k:
                res = min(res, i)

        if res == inf:
            return -1
        return res
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(n)`
