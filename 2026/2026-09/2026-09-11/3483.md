```python
class Solution:
    def totalNumbers(self, digits: List[int]) -> int:
        res = 0
        pool = digits
        cnt = 0
        seen = set()

        while cnt != len(digits):
            first = pool.pop()
            if not first:
                pool.insert(0, first)
                cnt += 1
                continue

            for i in range(len(pool)):
                second = pool[i]
                for j in range(i + 1, len(pool)):
                    third = pool[j]
                    if third % 2 == 0 and (first, second, third) not in seen:
                        res += 1
                        seen.add((first, second, third))

            for i in range(len(pool) -1, -1, -1):
                second = pool[i]
                for j in range(i + -1, -1, -1):
                    third = pool[j]
                    if third % 2 == 0 and (first, second, third) not in seen:
                        res += 1
                        seen.add((first, second, third))

            pool.insert(0, first)
            cnt += 1

        return res
```

> Time Complexity : `O(n^3)`
>
> Space Complexity : `O(n^3)`
