```python
class Solution:
    def firstStableIndex(self, nums: list[int], k: int) -> int:
        n = len(nums)
        max_min = [[] * i for i in range(n)] # [max_val, min_val] pairs

        curr_max = -inf
        for i in range(n):
            curr_max = max(curr_max, nums[i])
            max_min[i].append(curr_max)

        curr_min = inf
        for i in range(n - 1, -1, -1):
            curr_min = min(curr_min, nums[i])
            max_min[i].append(curr_min)

        res = inf
        for i in range(n):
            score = max_min[i][0] - max_min[i][1]
            if score <= k:
                res = min(res, i)

        if res == inf:
            return -1
        return res
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(n)`
