Tough question.

```python
class Node:
    def __init__(self, val, idx):
        self.val = val
        self.idx = idx
        self.prev = None
        self.next = None
        self.ver = 0   # version counter

class Solution:
    def minimumPairRemoval(self, nums):
        n = len(nums)
        head = Node(nums[0], 0)
        cur = head

        nodes = [head]
        decrease = 0

        for i in range(1, n):
            node = Node(nums[i], i)
            cur.next = node
            node.prev = cur
            if cur.val > node.val:
                decrease += 1
            cur = node
            nodes.append(node)

        pq = []

        def push(a, b):
            heapq.heappush(
                pq,
                (a.val + b.val, a.idx, a, b, a.ver, b.ver)
            )

        for i in range(n - 1):
            push(nodes[i], nodes[i + 1])

        ans = 0

        while decrease > 0:
            cost, _, a, b, va, vb = heapq.heappop(pq)

            # stale pair
            if a.ver != va or b.ver != vb or a.next != b:
                continue

            ans += 1

            if a.val > b.val:
                decrease -= 1

            prev = a.prev
            nxt = b.next

            # update decrease count
            if prev:
                if prev.val > a.val:
                    decrease -= 1
                if prev.val > cost:
                    decrease += 1

            if nxt:
                if b.val > nxt.val:
                    decrease -= 1
                if cost > nxt.val:
                    decrease += 1

            # merge
            a.val = cost
            a.ver += 1
            a.next = nxt
            if nxt:
                nxt.prev = a

            b.ver += 1  # invalidate b

            if prev:
                push(prev, a)
            if nxt:
                push(a, nxt)

        return ans
```

> Time Complexity : `O(n * log n)`
> 
> Space Complexity : `O(n)`
