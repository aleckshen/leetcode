The solution below is my first approach which is to brute all possible sizes of squares. If the sum of all cells within the current square is lesser than or equal to the threshold then we can update our maximum side length variable.

```python
class Solution(object):
    def maxSideLength(self, mat, threshold):
        ROWS = len(mat)
        COLS = len(mat[0])
        res = 0

        for size in range(min(ROWS, COLS) + 1):
            for r in range(ROWS - size + 1):
                for c in range(COLS - size + 1):
                    if self.isValid(mat, size, r, c, threshold):
                        res = max(res, size)

        return res

    def isValid(self, grid, size, r, c, threshold):
        total = 0

        for i in range(r, r + size):
            for j in range(c, c + size):
                total += grid[i][j]
                if total > threshold:
                    return False

        return True
```

This solution ended up exceeding the time limit. To optimize this solution we can store the prefix sum of all grids in a 2D array `prefix`. When we iterate through each square size we can use this prefix array to calculate the total sum in each square instead of recalculating cells.

```python
class Solution(object):
    def maxSideLength(self, mat, threshold):
        ROWS = len(mat)
        COLS = len(mat[0])
        res = 0

        prefix = [[0] * (COLS + 1) for _ in range(ROWS + 1)]
        for r in range(ROWS):
            for c in range(COLS):
                prefix[r + 1][c + 1] = mat[r][c] + prefix[r + 1][c] + prefix[r][c + 1] - prefix[r][c]

        for size in range(min(ROWS, COLS) + 1):
            for r in range(ROWS - size + 1):
                for c in range(COLS - size + 1):
                    total = (prefix[r + size][c + size] - prefix[r][c + size] - prefix[r + size][c] + prefix[r][c])
                    
                    if total <= threshold:
                        res = max(size, res)

        return res
```

> Time Complexity : `O(min(m, n) * m * n)`
> 
> Space Complexity : `O(m * n)`
